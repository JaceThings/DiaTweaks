cmake_minimum_required(VERSION 3.25)
project(DiaTweaksWindows VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The first prototype requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /permissive- /EHsc /utf-8)
include(cmake/catalog.cmake)
add_library(tap_runtime STATIC tap/session.cpp tap/tree.cpp tap/visibility.cpp tap/inspect.cpp tap/features.cpp)
target_include_directories(tap_runtime PUBLIC include tap)
target_link_libraries(tap_runtime PUBLIC ole32 oleaut32 user32 advapi32)
set(DIATWEAKS_WINUI_PROJECTIONS "" CACHE PATH "Generated C++/WinRT headers for the installed Dia WinUI metadata")
if(DIATWEAKS_WINUI_PROJECTIONS)
  if(NOT EXISTS "${DIATWEAKS_WINUI_PROJECTIONS}/winrt/Microsoft.UI.Xaml.Controls.h")
    message(FATAL_ERROR "WinUI projections are missing; see build-settings.ps1")
  endif()
  target_sources(tap_runtime PRIVATE settings/native_settings.cpp settings/page.cpp settings/entry_style.cpp settings/motion.cpp settings/content_host.cpp settings/selection_visual.cpp)
  target_sources(tap_runtime PRIVATE features/native_features.cpp features/layout_inspect.cpp)
  target_sources(tap_runtime PRIVATE features/layout_features.cpp features/toolbar_adapter.cpp)
  target_sources(tap_runtime PRIVATE features/window_appearance.cpp)
  target_sources(tap_runtime PRIVATE features/auto_settings.cpp)
  target_include_directories(tap_runtime PUBLIC settings features "${DIATWEAKS_WINUI_PROJECTIONS}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_precompile_headers(tap_runtime PRIVATE features/winui_pch.h)
  target_compile_definitions(tap_runtime PUBLIC DIATWEAKS_SETTINGS)
  target_link_libraries(tap_runtime PUBLIC windowsapp)
endif()
add_library(patch_preferences STATIC launcher/preferences.cpp launcher/preferences_sync.cpp)
target_include_directories(patch_preferences PUBLIC launcher include)
target_link_libraries(patch_preferences PUBLIC shell32 ole32 uuid)
if(DIATWEAKS_WINUI_PROJECTIONS)
  add_executable(DiaTweaks WIN32 launcher/main.cpp launcher/runner.cpp controller/target.cpp controller/channel.cpp)
  target_include_directories(DiaTweaks PRIVATE controller)
  target_link_libraries(DiaTweaks PRIVATE patch_preferences user32 advapi32)
endif()
add_library(DiaTweaksTap SHARED tap/exports.cpp tap/exports.def)
target_link_libraries(DiaTweaksTap PRIVATE tap_runtime)
add_executable(diatweaks-probe controller/main.cpp controller/target.cpp controller/channel.cpp)
target_include_directories(diatweaks-probe PRIVATE include controller)
target_link_libraries(diatweaks-probe PRIVATE ole32 advapi32 user32)
include(CTest)
if(BUILD_TESTING)
  add_executable(preferences-tests tests/preferences_tests.cpp)
  target_link_libraries(preferences-tests PRIVATE patch_preferences)
  add_test(NAME preferences-tests COMMAND preferences-tests)
  add_executable(prototype-tests tests/prototype_tests.cpp controller/channel.cpp)
  target_include_directories(prototype-tests PRIVATE controller)
  target_link_libraries(prototype-tests PRIVATE tap_runtime)
  add_dependencies(prototype-tests DiaTweaksTap)
  add_test(NAME prototype-tests COMMAND prototype-tests)
  add_test(NAME prototype-background COMMAND prototype-tests --background)
  set_tests_properties(prototype-tests PROPERTIES TIMEOUT 30)
  add_test(NAME controller-help COMMAND diatweaks-probe --help)
endif()
